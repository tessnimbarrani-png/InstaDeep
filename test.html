`; container.appendChild(div); } // Initial example projects addProject("AI Healthcare", 25, 2, 50000, 300000); addProject("AI Logistics", 18, 3, 50000, 250000); addProject("AI Finance", 30, 4, 50000, 300000); // Budget optimization function optimizeBudget() { const company = document.getElementById("company").value.trim(); const budget = Number(document.getElementById("budget").value); const currency = document.getElementById("currency").value; const penalty = Number(document.getElementById("riskPenalty").value); if (!company || !Number.isFinite(budget) || budget <= 0) { alert("Please enter a valid company name and budget."); return; } const rows = [...document.querySelectorAll(".project")]; if (rows.length === 0) { alert("Please add at least one project."); return; } const projects = rows.map(row => ({ name: row.querySelector(".p-name").value.trim(), roi: Number(row.querySelector(".p-roi").value), risk: Number(row.querySelector(".p-risk").value), min: Number(row.querySelector(".p-min").value), max: Number(row.querySelector(".p-max").value) })); for (const p of projects) { if ( !p.name || !Number.isFinite(p.roi) || p.roi < 0 || !Number.isInteger(p.risk) || p.risk < 1 || p.risk > 5 || !Number.isFinite(p.min) || p.min < 0 || !Number.isFinite(p.max) || p.max < p.min ) { alert("Please verify all project information."); return; } } const names = projects.map(p => p.name.toLowerCase()); if (new Set(names).size !== names.length) { alert("Project names must be unique."); return; } const minTotal = projects.reduce((sum, p) => sum + p.min, 0); if (minTotal > budget) { alert("Your budget is insufficient for the minimum investments."); return; } // Risk-adjusted ROI projects.forEach(p => { p.adjustedROI = p.roi - penalty * p.risk; p.allocation = p.min; }); // Sort projects by adjusted return const ranked = [...projects].sort( (a, b) => b.adjustedROI - a.adjustedROI ); // Allocate the remaining budget to the best projects let remaining = budget - minTotal; for (const p of ranked) { const available = p.max - p.min; const allocation = Math.min(remaining, available); p.allocation += allocation; remaining -= allocation; if (remaining <= 0.000001) break; } const invested = projects.reduce( (sum, p) => sum + p.allocation, 0 ); const profit = projects.reduce( (sum, p) => sum + p.allocation * p.roi / 100, 0 ); const remainingBudget = budget - invested; projects.forEach(p => { p.expectedProfit = p.allocation * p.roi / 100; }); lastResults = projects.map(p => ({...p})); displayResults( company, budget, invested, profit, remainingBudget, currency, projects ); } // Display financial results function displayResults( company, budget, invested, profit, remaining, currency, projects ) { document.getElementById("results").classList.remove("hidden"); document.getElementById("companyTitle").textContent = "Financial Report: " + company; const money = value => value.toLocaleString("en-US", { maximumFractionDigits: 2 }) + " " + currency; document.getElementById("available").textContent = money(budget); document.getElementById("invested").textContent = money(invested); document.getElementById("profit").textContent = money(profit); document.getElementById("remaining").textContent = money(remaining); const table = document.getElementById("resultTable"); table.innerHTML = ""; projects.forEach(p => { const tr = document.createElement("tr"); const values = [ p.name, money(p.allocation), p.roi + "%", p.risk + "/5", money(p.expectedProfit) ]; values.forEach(value => { const td = document.createElement("td"); td.textContent = value; tr.appendChild(td); }); table.appendChild(tr); }); // Budget chart if (budgetChart) { budgetChart.destroy(); } budgetChart = new Chart( document.getElementById("budgetChart"), { type: "doughnut", data: { labels: projects.map(p => p.name), datasets: [{ data: projects.map(p => p.allocation) }] }, options: { responsive: true, plugins: { legend: { position: "bottom" } } } } ); generateRecommendations(projects, remaining); } // Generate decision recommendations function generateRecommendations(projects, remaining) { const container = document.getElementById("recommendations"); container.innerHTML = ""; const best = [...projects].sort( (a, b) => b.adjustedROI - a.adjustedROI )[0]; const recommendations = [ "🏆 Best risk-adjusted project: " + best.name, "📊 Recommended investment: " + best.allocation.toLocaleString("en-US"), "💰 Expected profit: " + projects.reduce((s, p) => s + p.expectedProfit, 0) .toLocaleString("en-US"), "⚖️ Review projects with high risk before investing." ]; if (remaining > 0.01) { recommendations.push( "💡 Unallocated budget: " + remaining.toLocaleString("en-US") + ". Consider keeping it as a financial reserve." ); } recommendations.forEach(text => { const p = document.createElement("p"); p.textContent = text; p.style.margin = "12px 0"; container.appendChild(p); }); } // Export financial report function downloadReport() { if (!lastResults.length) { alert("Please optimize the budget first."); return; } const headers = [ "Project", "Investment", "ROI", "Risk", "Expected Profit", "Risk Adjusted ROI" ]; const rows = lastResults.map(p => [ p.name, p.allocation, p.roi, p.risk, p.expectedProfit, p.adjustedROI ]); const csv = [ headers, ...rows ].map(row => row.map(value => '"' + String(value).replace(/"/g, '""') + '"' ).join(",") ).join("\n"); const blob = new Blob( ["\uFEFF" + csv], {type: "text/csv;charset=utf-8;"} ); const link = document.createElement("a"); link.href = URL.createObjectURL(blob); link.download = "InstaDeep_Financial_Report.csv"; link.click(); URL.revokeObjectURL(link.href); }
